// Exercice 1

CREATE TABLE COMPETENCE(
	code_comp VARCHAR(10) PRIMARY KEY,
	libel_comp VARCHAR(40) NOT NULL
);

CREATE TABLE EMPLOYE(
	matricule_emp VARCHAR(10) PRIMARY KEY,
	nomp_emp VARCHAR(50) NOT NULL,
	prenom_emp VARCHAR(50) NOT NULL
);

CREATE TABLE possede(
	code_comp VARCHAR(10),
	matricule_emp VARCHAR(10),

	CONSTRAINT possede_pk PRIMARY KEY (code_comp, matricule_emp),
	CONSTRAINT possede_fk1 FOREIGN KEY (code_comp) REFERENCES COMPETENCE(code_comp),
	CONSTRAINT possede_fk2 FOREIGN KEY (matricule_emp) REFERENCES EMPLOYE(matricule_emp)
);

CREATE TABLE AFFAIRE(
	no_affaire INT PRIMARY KEY
);

CREATE TABLE intervenir(
	code_comp VARCHAR(10),
	matricule_emp VARCHAR(10),
	no_affaire INT,

	CONSTRAINT intervenir_pk PRIMARY KEY (code_comp, matricule_emp, no_affaire),
	CONSTRAINT intervenir_fk1 FOREIGN KEY (code_comp, matricule_emp) REFERENCES possede(code_comp, matricule_emp),
	CONSTRAINT intervenir_fk1 FOREIGN KEY (no_affaire) REFERENCES AFFAIRE(no_affaire)
);

// Exercice 2

VAR MAISON REAL RELATION{
	no_proj_con CHARACTER,
	typ_maison CHARACTER,
	surface INT} KEY {no_proj_con};

VAR ENTREPRISE REAL RELATION{
	code_ent CHARACTER,
	nom_ent CHARACTER,
	adresse_ent CHARACTER,
	tel_ent CHARACTER} KEY {code_ent};

VAR LOT_TRAVAUX REAL RELATION{
	no_lot CHARACTER,
	libelle_lot CHARACTER} KEY {no_lot};

VAR offres REAL RELATION{
	no_proj_con CHARACTER,
	code_ent CHARACTER,
	no_lot CHARACTER} KEY {no_proj_con, code_ent, no_lot};
CONSTRAINT offres_FK1 offres {no_proj_con} <= MAISON {no_proj_con};
CONSTRAINT offres_FK2 offres {code_ent} <= ENTREPRISE {code_ent};
CONSTRAINT offres_FK3 offres {no_lot} <= LOT_TRAVAUX {no_lot};

VAR attribution REAL RELATION{
	no_proj_con CHARACTER,
	code_ent CHARACTER,
	no_lot CHARACTER} KEY {no_proj_con, no_lot};
CONSTRAINT attribution_FK1 attribution <= offres;
